\section{PART 1 --- FTP Protocol Analysis (Phân tích giao thức FTP)}

\subsection{Tổng quan \& Môi trường thử nghiệm}
Trong phần này, nhóm thực hiện bắt và phân tích gói tin của giao thức FTP (File Transfer Protocol) trên 2 môi trường:
\begin{itemize}
  \item \textbf{Phiên 1 (Public FTP Server):} Máy local (IP: \texttt{192.168.1.26}) kết nối đến Public FTP Server \texttt{ftp.dlptest.com} (IP: \texttt{44.241.66.173}) qua phần mềm Command Prompt (CMD) và Wireshark.
  \item \textbf{Phiên 2 (Project 01 FTP):} Kết nối giữa ứng dụng FTP Client và FTP Server tự phát triển từ Project 01, chạy thực tế trên 2 máy PC riêng biệt cùng mạng LAN.
\end{itemize}

\subsection{Phân tích phiên 1: Public FTP Test Server (ftp.dlptest.com)}
 \subsubsection{Thí nghiệm 1: Khởi tạo kết nối FTP (Establish Connection)}
  \begin{itemize}
    \item \textbf{Cổng sử dụng (Ports):} Cổng điều khiển phía Server là \texttt{21}. Cổng nguồn phía Client là \texttt{57690} (được hệ điều hành cấp phát ngẫu nhiên - Ephemeral Port).
    \item \textbf{Địa chỉ IP của Server:} \texttt{44.241.66.173}.
    \item \textbf{Gói tin thiết lập kết nối:} Quá trình bắt tay 3 bước TCP (TCP 3-Way Handshake) bắt đầu tại gói tin số \texttt{638} (\texttt{[SYN]}), tiếp theo là gói \texttt{673} (\texttt{[SYN, ACK]}) và hoàn tất tại gói \texttt{674} (\texttt{[ACK]}).
    \item \textbf{Giao thức xuất hiện trước FTP:} Giao thức \texttt{TCP} ở tầng Transport phải thiết lập kết nối thành công trước khi các lệnh FTP ở tầng Application có thể gửi đi.
  \end{itemize}

  \subsubsection{Thí nghiệm 2: Xác thực đăng nhập (Login Authentication)}
  \begin{itemize}
    \item \textbf{Kiểm tra Username \& Password:} Có thể nhìn thấy trực tiếp Username (\texttt{USER dlpuser}) và Password (\texttt{PASS rNrKYTX9g7z3RgJRmxWuGHbeu}) hiển thị dưới dạng văn bản rõ (Cleartext / Plaintext) trong gói tin FTP. Tuy nhiên, trong quá trình đăng nhập sẽ không thể thấy những gì ta gõ vào khi nhập Password, vì Client sẽ gửi trực tiếp gói tin \texttt{PASS} chứa mật khẩu mà không hiển thị trên màn hình.
    \begin{figure}[H]
      \centering

    \item \textbf{Mức độ nguy hiểm:} Do FTP truyền thông tin dạng rõ không mã hóa, kẻ tấn công trên cùng mạng (Eavesdropper) có thể dùng các công cụ bắt gói tin (như Wireshark, Ettercap) để đánh cắp tài khoản đăng nhập (Sniffing Attack).
    \item \textbf{Mã phản hồi đăng nhập thành công:} Server trả về gói tin chứa mã phản hồi \texttt{230 User logged in, proceed}.
  \end{itemize}

  \subsubsection{Thí nghiệm 3: Xem danh sách thư mục (Directory Listing)}
  \begin{itemize}
    \item \textbf{Lệnh yêu cầu danh sách thư mục:} Lệnh thô \texttt{LIST} (tương đương với \texttt{dir} trên CLI) được Client gửi tới Server.
    \item \textbf{Cổng truyền dữ liệu (Data Port):} Do kết nối hoạt động ở Active Mode (thông qua lệnh PORT 192,168,1,26,202,34), cổng nhận dữ liệu phía Client được tính là:$$Port = (202 \times 256) + 34 = \mathbf{51746}$$ Dữ liệu được truyền giữa cổng 20 phía Server và cổng 51746 phía Client.
    \item \textbf{Kết nối TCP truyền danh sách:} Là kết nối TCP Data riêng biệt được khởi tạo giữa Client (192\.168\.1\.26\:51746) và Server (44\.241\.66\.173\:20), hoạt động độc lập với kết nối TCP Control (Port 21).
  \end{itemize}

    \item \textbf{Thời gian truyền (Duration):} Tính bằng hiệu thời gian giữa gói tin khởi tạo kết nối dữ liệu và gói phản hồi \texttt{226 Transfer complete} từ Server.

  \subsubsection{Thí nghiệm 5: Download File (Download a File)}
  \begin{itemize}
    \item \textbf{Lệnh Download file:} Lệnh thô \texttt{RETR test.txt}.
    \item \textbf{Số gói tin \& Kích thước file:} Tổng số gói tin TCP Data truyền qua mạng và dung lượng file tính theo byte.
    \item \textbf{TCP Stream chứa file:} Stream TCP riêng biệt tương ứng với kết nối dữ liệu.
    \item \textbf{Active hay Passive Mode?:} Trong phiên làm việc bằng CMD trên Windows, ứng dụng sử dụng \textbf{Active Mode}. Minh chứng là gói tin lệnh \texttt{PORT} được Client gửi tới Server để yêu cầu Server chủ động kết nối về phía Client.

  So sánh điểm tương đồng và khác biệt giữa Server thực tế trên Internet và Server tự cài đặt trên mạng nội bộ.

\section{PART 2 --- HTTP Protocol Analysis (Phân tích giao thức HTTP)}

\subsection{Chương trình Socket Client (Part A --- Socket Client Code)}
  \subsubsection{Mã nguồn chương trình (\texttt{http\_client.py} / \texttt{http\_client.c})}
  Trình bày đoạn mã nguồn lập trình Socket thô kết nối đến cổng 80 của target host.

  \subsubsection{Giải thích chi tiết các bước xử lý trong Socket Code}
  Giải thích chi tiết từng lệnh khởi tạo socket, thiết lập kết nối TCP, tạo chuỗi HTTP GET request và đọc dữ liệu response.

\subsection{Bắt gói tin Wireshark (Part B --- Packet Capture)}
  Trình bày cấu hình capture filter \texttt{tcp port 80} và quá trình lưu file \texttt{lab\_capture.pcapng}.

\subsection{Trả lời các câu hỏi phân tích chi tiết (Part C --- Analysis Questions)}

  \subsubsection{A. Name Resolution \& Addressing (Phân giải tên miền \& Địa chỉ)}
  Trả lời Câu 1 -- 2 về truy vấn DNS và địa chỉ IP kết nối.

  \subsubsection{B. The TCP Connection (Kết nối TCP)}
  Trả lời Câu 3 -- 6 về bắt tay 3 bước TCP, Source Port, thông số MSS, Window Size và thời gian RTT.

  \subsubsection{C. The HTTP Request (Yêu cầu HTTP)}
  Trả lời Câu 7 -- 9 về phương thức HTTP GET, các dòng Header và dung lượng Request.

  \subsubsection{D. The HTTP Response (Phản hồi HTTP)}
  Trả lời Câu 10 -- 14 về Status Code (301 Redirect), Location header, Response headers và dấu hiệu kết thúc Message Body.

  \subsubsection{E. TCP Segmentation \& Reassembly (Phân đoạn \& Tái hợp dữ liệu TCP)}
  Trả lời Câu 15 -- 20 về cơ chế phân đoạn TCP theo MSS, Sequence Number, ACK Number và quá trình tái hợp dữ liệu của Wireshark.

  \subsubsection{F. Closing the Connection \& Totals (Đóng kết nối \& Tổng kết dữ liệu)}
  Trả lời Câu 21 -- 22 về các gói tin FIN/RST ngắt kết nối và so sánh tổng lượng byte gửi/nhận.

  \subsubsection{G. Reflection (Đánh giá \& Nhìn nhận)}
  Trả lời Câu 23 -- 24 về vấn đề bảo mật của HTTP (Port 80) vs HTTPS (Port 443) và tóm tắt vòng đời đầy đủ của một HTTP Request.
