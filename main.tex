\subsection{Thảo luận \& Đánh giá mở rộng (Discussion \& Analysis)}
\subsubsection{Phân tích về lệnh và mã phản hổi FTP}
\begin{itemize}
  \item \textbf{Cấu trúc câu lệnh (Client Commands):}
    \begin{itemize}
      \item FTP là giao thức dạnng văn bản (Text-based/ ASCII).
      \item Mỗi câu lệnh từ Client gửi lên Server có cú pháp: \texttt{<COMMAND> [<parameter>] \textbackslash r \textbackslash n}.
      \item VD: USER dlpuser\texttt{\textbackslash r\textbackslash n}
    \end{itemize}
  \item \textbf{Cấu trúc mã phản hồi (Server responds):}
    Mọi phản hồi của FTP Server đều gồm 3 chữ số + thông điệp mô tả:
    \begin{itemize}
      \item Chữ số đầu tiên (Ý nghĩa trạng thái chung:)
        \begin{itemize}
          \item \texttt{1xx} (Positive Preliminary): Đã chấp nhận lệnh, đang mở kết nối dữ liệu (ví dụ: 150 Opening BINARY mode data connection).
          \item \texttt{2xx} (Positive Completion): Lệnh hoàn thành thành công (ví dụ: 200 Command okay, 230 User logged in, 226 Transfer complete).
          \item \texttt{3xx} (Positive Intermediate): Lệnh được chấp nhận nhưng cần thêm thông tin (ví dụ: 331 Please specify the password).
          \item \texttt{4xx} (Transient Negative): Lỗi tạm thời, có thể thử lại sau (ví dụ: 425 Can't open data connection).
          \item \texttt{5xx} (Permanent Negative): Lỗi vĩnh viễn (ví dụ: 530 Not logged in, 550 File not found).
        \end{itemize}
      \item Chữ số thứ hai (Phân loại chức năng):
        \begin{itemize}
          \item \texttt{x0x}: Lỗi cú pháp (Syntax).
          \item \texttt{x1x}: Thông tin hệ thống (Information).
          \item \texttt{x2x}: Kết nối điều khiển \& dữ liệu (Connections).
          \item \texttt{x3x}: Xác thực \& Quyền hạn (Authentication).
          \item \texttt{x5x}: Hệ thống tập tin (File system).
        \end{itemize}
    \end{itemize}
\end{itemize}

\subsubsection{So sánh chi tiết Active Mode vs Passive Mode}
\begin{itemize}
  \item \textbf{Active Mode (chế độ chủ động):} Clien gửi lệnh \textbf{PORT} chứa địa chỉ IP và cổng nguồn của Client tới Server. Server sẽ là bên \textbf{chủ động khởi tạo} kết nối dữ liệu từ cổng 20 đến cổng của Client. Nhược điểm là dễ bị Firewall/Nat bên phía Clinet chặn kết nối đến.
  \item \textbf{Passive Mode (chế độ thụ động):} Client gửi lệnh \textbf{PASV} tới Server. Server sẽ phản hồi mã \texttt{227}, sau đó \textbf{chủ động mở cổng ngẫu nhiên} và trả về địa chỉ IP + cổng cho Client. Client sẽ là bên \textbf{chủ động khởi tạo} kết nối dữ liệu đến Server. Ưu điểm là dễ dàng vượt qua Firewall/Nat của Client.
\end{itemize}

\subsubsection{Đánh giá an toàn thông tin đối với FTP}
Giao thức FTP gốc truyền thông tin dạng rõ (plaintext) không mã hóa, do đó dễ bị tấn công đánh cắp thông tin đăng nhập (Sniffing Attack). Để bảo mật hơn, nên sử dụng các giao thức FTP bảo mật như FTPS (FTP over SSL/TLS) hoặc SFTP (SSH File Transfer Protocol) để mã hóa toàn bộ luồng truyền.

\section{PART 2 --- HTTP Protocol Analysis (Phân tích giao thức HTTP)}

\subsection{Part A --- Socket Client (Python)}
File: \texttt{http\_client.py}

\begin{lstlisting}[style=pystyle,language=Python]
import socket

host = "www.google.com"
port = 80

# Create a TCP socket over IPv4 and connect to the web server
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.connect((host, port))

# Build and send a minimal HTTP/1.1 GET request
request = (
    "GET / HTTP/1.1\r\n"
    "Host: www.google.com\r\n"
    "User-Agent: SocketLab/1.0\r\n"
    "Connection: close\r\n"
    "\r\n"
)
s.sendall(request.encode())

# Read the response until the server closes the connection
response = b""
while True:
    chunk = s.recv(4096)
    if not chunk:
        break
    response += chunk

print(response.decode(errors="replace"))
s.close()
\end{lstlisting}
